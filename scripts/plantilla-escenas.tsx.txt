import React from "react";
import { AbsoluteFill, interpolate, useCurrentFrame } from "remotion";
import { Badge } from "../../kit/Badge";
import { Camera, type CamKey } from "../../kit/Camera";
import { REAL, sombreado } from "../../kit/colores";
import { Label } from "../../kit/Label";
import { GlowDefs, Pulse } from "../../kit/Pulse";
import { Sfx } from "../../kit/Sfx";
import { useCues, useScene } from "../../scene-context";

/**
 * Escenas de "__SLUG__" (registro __ESTILO__).
 * 1) Haz el storyboard: por cada frase, qué se VE y qué palabra lo dispara.
 * 2) Dibuja el "mundo" (SVG 1080×1920 o más grande) con colores REALES y sombreado().
 * 3) Sincroniza con useCues([...palabras de la narración]).
 * 4) Lo que suena se escucha con <Sfx>, no se escribe.
 * 5) Marca el arte clave con data-qa="importante" y la utilería móvil con data-qa="objeto";
 *    `npm run qa` avisará si algo lo tapa. Las <Badge> van sin x/y (franja fija).
 * Reemplaza los marcadores de abajo por la ilustración real.
 */

const k = (at: number, x: number, y: number, zoom: number): CamKey => ({ at, shot: { x, y, zoom } });

/** Lienzo con cámara continua: mundo SVG de 1080×1920 (amplíalo si la cámara viaja). */
const Lienzo: React.FC<{ keys: CamKey[]; children: React.ReactNode }> = ({ keys, children }) => (
  <AbsoluteFill>
    <Camera keys={keys} worldWidth={1080} worldHeight={1920}>
      <svg width={1080} height={1920} viewBox="0 0 1080 1920" style={{ overflow: "visible" }}>
        <GlowDefs />
        {children}
      </svg>
    </Camera>
  </AbsoluteFill>
);

/** Marcador temporal: un objeto con color real que aparece en su cue. Reemplázalo. */
const Marcador: React.FC<{ x: number; y: number; at: number; color: string; texto: string }> = ({ x, y, at, color, texto }) => {
  const f = useCurrentFrame();
  const s = sombreado(color);
  const p = interpolate(f, [at, at + 10], [0, 1], { extrapolateLeft: "clamp", extrapolateRight: "clamp" });
  return (
    <g data-qa="importante" data-qa-nombre={texto} transform={`translate(${x} ${y}) scale(${p})`}>
      <circle r={150} fill={s.base} stroke={s.contorno} strokeWidth={8} />
      <circle cx={-50} cy={-50} r={40} fill={s.brillo} opacity={0.6} />
      <text textAnchor="middle" y={12} fontFamily="Inter" fontWeight={800} fontSize={40} fill="#fff">
        {texto}
      </text>
    </g>
  );
};

export const Gancho: React.FC = () => {
  const { durationInFrames } = useScene();
  const [a] = useCues(["__CUE1__"]);
  return (
    <>
      <Sfx name="whoosh" at={a} dur={20} volume={0.4} />
      <Lienzo keys={[k(0, 540, 1080, 1), k(durationInFrames, 540, 1050, 1.15)]}>
        <Marcador x={540} y={1000} at={a} color={REAL.cuerpo.musculo} texto="GANCHO" />
        <Label x={540} y={1000} tx={760} ty={760} text="Etiqueta" at={a + 10} />
      </Lienzo>
    </>
  );
};

export const Mecanismo: React.FC = () => {
  const { durationInFrames, audioStartFrame, audioFrames } = useScene();
  const ruta = "M200 1300 C400 900 700 1400 880 800";
  return (
    <Lienzo keys={[k(0, 540, 1080, 1.1), k(durationInFrames, 600, 1000, 1.3)]}>
      <Pulse d={ruta} start={audioStartFrame} duration={Math.max(20, audioFrames * 0.6)} />
      <Marcador x={880} y={800} at={audioStartFrame + audioFrames * 0.6} color={REAL.naturaleza.agua} texto="MECANISMO" />
    </Lienzo>
  );
};

export const Detalle: React.FC = () => {
  const { durationInFrames, audioStartFrame } = useScene();
  return (
    <>
      <Lienzo keys={[k(0, 540, 1080, 0.9), k(durationInFrames, 540, 1080, 1.2)]}>
        <Marcador x={540} y={1050} at={4} color={REAL.cuerpo.cerebro} texto="DETALLE" />
      </Lienzo>
      <Badge text="Dato con fuente" icono="info" at={audioStartFrame + 10} />
    </>
  );
};

export const Cierre: React.FC<{ fuente?: string }> = ({ fuente }) => {
  const { durationInFrames } = useScene();
  return (
    <>
      <Lienzo keys={[k(0, 540, 1050, 1.15), k(durationInFrames, 540, 1080, 1)]}>
        <Marcador x={540} y={1000} at={2} color={REAL.cuerpo.musculo} texto="CIERRE" />
      </Lienzo>
      {fuente ? (
        <div style={{ position: "absolute", left: 80, right: 160, bottom: 300, fontFamily: "monospace", fontSize: 22, color: "#ddd" }}>
          FUENTE · {fuente}
        </div>
      ) : null}
    </>
  );
};
